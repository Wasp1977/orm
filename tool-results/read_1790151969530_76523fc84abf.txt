   711→
   712→  /* operator cost calculations */
   713→  const kitIncludedOps = selectedKit ? selectedKit.operatorCount : 0;
   714→  const selectedOpCount = checkedEmps.size > 0 ? checkedEmps.size : kitIncludedOps;
   715→  const extraOps = Math.max(0, selectedOpCount - kitIncludedOps);
   716→  const extraOpsCost = extraOps * ADDITIONAL_OPERATOR_PRICE;
   717→
   718→  const toggleEmp = (idx: number) => {
   719→    setCheckedEmps((prev) => {
   720→      const next = new Set(prev);
   721→      if (next.has(idx)) next.delete(idx);
   722→      else next.add(idx);
   723→      return next;
   724→    });
